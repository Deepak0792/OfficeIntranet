-- =============================================
-- EMPLOYEE DIRECTORY / GLOBAL HR PLATFORM
-- ENTERPRISE DATABASE DESIGN FOR SQL SERVER
-- =============================================

-- =============================================
-- DESIGNATION MASTER
-- Stores employee designations/titles
-- =============================================
CREATE TABLE Designation (
    DesignationId              BIGINT PRIMARY KEY IDENTITY(1,1),
    DesignationCode            NVARCHAR(50) UNIQUE,
    DesignationName            NVARCHAR(200) NOT NULL,
    Grade                      NVARCHAR(50),
    IsActive                   BIT NOT NULL DEFAULT 1,
    CreatedAt                  DATETIME2 NOT NULL DEFAULT GETUTCDATE()
);


-- =============================================
-- TIME ZONE MASTER
-- Stores global time zone definitions
-- =============================================
CREATE TABLE TimeZoneMaster (
    TimeZoneId                 BIGINT PRIMARY KEY IDENTITY(1,1),
    TimeZoneCode               NVARCHAR(100) NOT NULL UNIQUE,
    TimeZoneName               NVARCHAR(200) NOT NULL,
    UtcOffset                  NVARCHAR(20) NOT NULL,
    OffsetMinutes              INT NOT NULL,
    SupportsDaylightSaving     BIT NOT NULL DEFAULT 0,
    WindowsTimeZoneId          NVARCHAR(200),
    IanaTimeZoneId             NVARCHAR(200),
    CountryCode                NVARCHAR(10) NULL,
    IsActive                   BIT NOT NULL DEFAULT 1,
    CreatedAt                  DATETIME2 NOT NULL DEFAULT GETUTCDATE()
);


-- =============================================
-- COUNTRY MASTER
-- Stores country information
-- =============================================
CREATE TABLE Country (
    CountryId                  BIGINT PRIMARY KEY IDENTITY(1,1),
    CountryCode                NVARCHAR(10) NOT NULL UNIQUE,
    CountryName                NVARCHAR(200) NOT NULL,
    CurrencyCode               NVARCHAR(10),
    TimeZoneId                 BIGINT,
    IsActive                   BIT NOT NULL DEFAULT 1,
    CreatedAt                  DATETIME2 NOT NULL DEFAULT GETUTCDATE()
);


-- =============================================
-- LEGAL ENTITY MASTER
-- Stores company legal entities/subsidiaries
-- =============================================
CREATE TABLE LegalEntity (
    LegalEntityId              BIGINT PRIMARY KEY IDENTITY(1,1),
    EntityCode                 NVARCHAR(50) UNIQUE,
    EntityName                 NVARCHAR(300) NOT NULL,
    CountryId                  BIGINT NOT NULL,
    TaxIdentificationNumber    NVARCHAR(100),
    RegistrationNumber         NVARCHAR(100),
    CurrencyCode               NVARCHAR(10),
    IsActive                   BIT NOT NULL DEFAULT 1,
    CreatedAt                  DATETIME2 NOT NULL DEFAULT GETUTCDATE(),

    CONSTRAINT FK_LegalEntity_Country
        FOREIGN KEY (CountryId)
        REFERENCES Country(CountryId)
);

-- =============================================
-- REGION MASTER
-- Stores region/state/city hierarchy
-- =============================================
CREATE TABLE Region (
    RegionId                   BIGINT PRIMARY KEY IDENTITY(1,1),
    CountryId                  BIGINT NOT NULL,
    RegionName                 NVARCHAR(200) NOT NULL,
    RegionType                 NVARCHAR(50),
    ParentRegionId             BIGINT NULL,
    IsActive                   BIT NOT NULL DEFAULT 1,
    CreatedAt                  DATETIME2 NOT NULL DEFAULT GETUTCDATE(),

    CONSTRAINT FK_Region_Country
        FOREIGN KEY (CountryId)
        REFERENCES Country(CountryId),

    CONSTRAINT FK_Region_Parent
        FOREIGN KEY (ParentRegionId)
        REFERENCES Region(RegionId)
);

-- =============================================
-- OFFICE LOCATION MASTER
-- Stores global office locations
-- =============================================
CREATE TABLE OfficeLocation (
    LocationId                 BIGINT PRIMARY KEY IDENTITY(1,1),
    LegalEntityId              BIGINT NOT NULL,
    CountryId                  BIGINT NOT NULL,
    RegionId                   BIGINT NULL,
    LocationCode               NVARCHAR(50) UNIQUE,
    LocationName               NVARCHAR(200) NOT NULL,
    BuildingName               NVARCHAR(200),
    AddressLine1               NVARCHAR(500),
    AddressLine2               NVARCHAR(500),
    City                       NVARCHAR(100),
    StateProvince              NVARCHAR(100),
    PostalCode                 NVARCHAR(20),
    Latitude                   DECIMAL(10,7),
    Longitude                  DECIMAL(10,7),
    TimeZoneId                 BIGINT,
    IsHeadOffice               BIT NOT NULL DEFAULT 0,
    IsActive                   BIT NOT NULL DEFAULT 1,
    CreatedAt                  DATETIME2 NOT NULL DEFAULT GETUTCDATE(),

    CONSTRAINT FK_OfficeLocation_LegalEntity
        FOREIGN KEY (LegalEntityId)
        REFERENCES LegalEntity(LegalEntityId),

    CONSTRAINT FK_OfficeLocation_Country
        FOREIGN KEY (CountryId)
        REFERENCES Country(CountryId),

    CONSTRAINT FK_OfficeLocation_Region
        FOREIGN KEY (RegionId)
        REFERENCES Region(RegionId)
);

-- =============================================
-- DEPARTMENT MASTER
-- Stores organization departments
-- =============================================
CREATE TABLE Department (
    DepartmentId               BIGINT PRIMARY KEY IDENTITY(1,1),
    DepartmentCode             NVARCHAR(50) UNIQUE,
    DepartmentName             NVARCHAR(200) NOT NULL,
    ParentDepartmentId         BIGINT NULL,
    Description                NVARCHAR(1000),
    IsActive                   BIT NOT NULL DEFAULT 1,
    CreatedAt                  DATETIME2 NOT NULL DEFAULT GETUTCDATE(),

    CONSTRAINT FK_Department_Parent
        FOREIGN KEY (ParentDepartmentId)
        REFERENCES Department(DepartmentId)
);

-- =============================================
-- EMPLOYEE MASTER
-- Stores employee core profile data
-- =============================================
CREATE TABLE Employee (
    EmployeeId                 BIGINT PRIMARY KEY IDENTITY(1,1),
    EmployeeCode               NVARCHAR(50) NOT NULL UNIQUE,
    FirstName                  NVARCHAR(100) NOT NULL,
    LastName                   NVARCHAR(100),
    DisplayName                NVARCHAR(200),
    Email                      NVARCHAR(255) NOT NULL UNIQUE,
    MobileNumber               NVARCHAR(30),
    DesignationId              BIGINT NULL,
    PreferredLanguage          NVARCHAR(20),
    PreferredTimeZoneId        BIGINT,
    DateOfJoining              DATE,
    EmploymentType             NVARCHAR(50),
    AboutMe                    NVARCHAR(MAX),
    ProfilePhotoUrl            NVARCHAR(1000),
    IsActive                   BIT NOT NULL DEFAULT 1,
    CreatedAt                  DATETIME2 NOT NULL DEFAULT GETUTCDATE(),
    UpdatedAt                  DATETIME2 NULL,

    CONSTRAINT FK_Employee_Designation
        FOREIGN KEY (DesignationId)
        REFERENCES Designation(DesignationId)
);

-- =============================================
-- EMPLOYEE LEGAL ENTITY MAPPING
-- Maps employees to legal entities
-- =============================================
CREATE TABLE EmployeeLegalEntity (
    EmployeeLegalEntityId      BIGINT PRIMARY KEY IDENTITY(1,1),
    EmployeeId                 BIGINT NOT NULL,
    LegalEntityId              BIGINT NOT NULL,
    IsPrimary                  BIT NOT NULL DEFAULT 0,
    StartDate                  DATE,
    EndDate                    DATE,
    IsActive                   BIT NOT NULL DEFAULT 1,

    CONSTRAINT FK_ELE_Employee
        FOREIGN KEY (EmployeeId)
        REFERENCES Employee(EmployeeId),

    CONSTRAINT FK_ELE_LegalEntity
        FOREIGN KEY (LegalEntityId)
        REFERENCES LegalEntity(LegalEntityId)
);

-- =============================================
-- EMPLOYEE DEPARTMENT MAPPING
-- Maps employees to departments
-- =============================================
CREATE TABLE EmployeeDepartment (
    EmployeeDepartmentId       BIGINT PRIMARY KEY IDENTITY(1,1),
    EmployeeId                 BIGINT NOT NULL,
    DepartmentId               BIGINT NOT NULL,
    IsPrimaryDepartment        BIT NOT NULL DEFAULT 0,
    AllocationPercentage       DECIMAL(5,2),
    StartDate                  DATE,
    EndDate                    DATE,
    IsActive                   BIT NOT NULL DEFAULT 1,

    CONSTRAINT FK_EmployeeDepartment_Employee
        FOREIGN KEY (EmployeeId)
        REFERENCES Employee(EmployeeId),

    CONSTRAINT FK_EmployeeDepartment_Department
        FOREIGN KEY (DepartmentId)
        REFERENCES Department(DepartmentId)
);

-- =============================================
-- EMPLOYEE LOCATION MAPPING
-- Maps employees to office locations
-- =============================================
CREATE TABLE EmployeeLocation (
    EmployeeLocationId         BIGINT PRIMARY KEY IDENTITY(1,1),
    EmployeeId                 BIGINT NOT NULL,
    LocationId                 BIGINT NOT NULL,
    IsPrimaryLocation          BIT NOT NULL DEFAULT 1,
    StartDate                  DATE,
    EndDate                    DATE,
    IsActive                   BIT NOT NULL DEFAULT 1,

    CONSTRAINT FK_EmployeeLocation_Employee
        FOREIGN KEY (EmployeeId)
        REFERENCES Employee(EmployeeId),

    CONSTRAINT FK_EmployeeLocation_Location
        FOREIGN KEY (LocationId)
        REFERENCES OfficeLocation(LocationId)
);

-- =============================================
-- RELATIONSHIP TYPE MASTER
-- Stores employee relationship types
-- =============================================
CREATE TABLE RelationshipType (
    RelationshipTypeId         BIGINT PRIMARY KEY IDENTITY(1,1),
    RelationshipName           NVARCHAR(100) NOT NULL UNIQUE,
    Description                NVARCHAR(500),
    IsActive                   BIT NOT NULL DEFAULT 1
);

-- =============================================
-- EMPLOYEE RELATIONSHIP GRAPH
-- Stores reporting and matrix hierarchy
-- =============================================
CREATE TABLE EmployeeRelationship (
    EmployeeRelationshipId     BIGINT PRIMARY KEY IDENTITY(1,1),
    ParentEmployeeId           BIGINT NOT NULL,
    ChildEmployeeId            BIGINT NOT NULL,
    RelationshipTypeId         BIGINT NOT NULL,
    DepartmentId               BIGINT NULL,
    IsPrimaryRelationship      BIT NOT NULL DEFAULT 0,
    EffectiveFrom              DATE,
    EffectiveTo                DATE,
    IsActive                   BIT NOT NULL DEFAULT 1,

    CONSTRAINT FK_ER_ParentEmployee
        FOREIGN KEY (ParentEmployeeId)
        REFERENCES Employee(EmployeeId),

    CONSTRAINT FK_ER_ChildEmployee
        FOREIGN KEY (ChildEmployeeId)
        REFERENCES Employee(EmployeeId),

    CONSTRAINT FK_ER_RelationshipType
        FOREIGN KEY (RelationshipTypeId)
        REFERENCES RelationshipType(RelationshipTypeId),

    CONSTRAINT FK_ER_Department
        FOREIGN KEY (DepartmentId)
        REFERENCES Department(DepartmentId)
);

-- =============================================
-- TEAM MASTER
-- Stores teams/squads/pods
-- =============================================
CREATE TABLE Team (
    TeamId                     BIGINT PRIMARY KEY IDENTITY(1,1),
    TeamCode                   NVARCHAR(50) UNIQUE,
    TeamName                   NVARCHAR(200) NOT NULL,
    TeamType                   NVARCHAR(100),
    Description                NVARCHAR(1000),
    IsActive                   BIT NOT NULL DEFAULT 1,
    CreatedAt                  DATETIME2 NOT NULL DEFAULT GETUTCDATE()
);

-- =============================================
-- EMPLOYEE TEAM MAPPING
-- Maps employees to teams
-- =============================================
CREATE TABLE EmployeeTeam (
    EmployeeTeamId             BIGINT PRIMARY KEY IDENTITY(1,1),
    EmployeeId                 BIGINT NOT NULL,
    TeamId                     BIGINT NOT NULL,
    RoleInTeam                 NVARCHAR(100),
    AllocationPercentage       DECIMAL(5,2),
    StartDate                  DATE,
    EndDate                    DATE,
    IsActive                   BIT NOT NULL DEFAULT 1,

    CONSTRAINT FK_EmployeeTeam_Employee
        FOREIGN KEY (EmployeeId)
        REFERENCES Employee(EmployeeId),

    CONSTRAINT FK_EmployeeTeam_Team
        FOREIGN KEY (TeamId)
        REFERENCES Team(TeamId)
);

-- =============================================
-- SKILL MASTER
-- Stores skill catalog
-- =============================================
CREATE TABLE Skill (
    SkillId                    BIGINT PRIMARY KEY IDENTITY(1,1),
    SkillName                  NVARCHAR(200) NOT NULL UNIQUE,
    SkillCategory              NVARCHAR(100),
    Description                NVARCHAR(1000),
    IsActive                   BIT NOT NULL DEFAULT 1
);

-- =============================================
-- EMPLOYEE SKILL MAPPING
-- Maps employees to skills
-- =============================================
CREATE TABLE EmployeeSkill (
    EmployeeSkillId            BIGINT PRIMARY KEY IDENTITY(1,1),
    EmployeeId                 BIGINT NOT NULL,
    SkillId                    BIGINT NOT NULL,
    SkillLevel                 NVARCHAR(50),
    YearsOfExperience          DECIMAL(5,2),
    IsPrimarySkill             BIT NOT NULL DEFAULT 0,
    LastUsedDate               DATE,

    CONSTRAINT FK_EmployeeSkill_Employee
        FOREIGN KEY (EmployeeId)
        REFERENCES Employee(EmployeeId),

    CONSTRAINT FK_EmployeeSkill_Skill
        FOREIGN KEY (SkillId)
        REFERENCES Skill(SkillId)
);

-- =============================================
-- EMPLOYEE CONTACTS
-- Stores employee contact methods
-- =============================================
CREATE TABLE EmployeeContact (
    EmployeeContactId          BIGINT PRIMARY KEY IDENTITY(1,1),
    EmployeeId                 BIGINT NOT NULL,
    ContactType                NVARCHAR(50),
    ContactValue               NVARCHAR(500),
    IsPrimary                  BIT NOT NULL DEFAULT 0,

    CONSTRAINT FK_EmployeeContact_Employee
        FOREIGN KEY (EmployeeId)
        REFERENCES Employee(EmployeeId)
);

-- =============================================
-- DOCUMENT TYPE MASTER
-- Stores employee document types
-- =============================================
CREATE TABLE DocumentType (
    DocumentTypeId             BIGINT PRIMARY KEY IDENTITY(1,1),
    DocumentTypeCode           NVARCHAR(50) UNIQUE,
    DocumentTypeName           NVARCHAR(200) NOT NULL,
    Description                NVARCHAR(1000),
    IsMandatory                BIT NOT NULL DEFAULT 0,
    IsActive                   BIT NOT NULL DEFAULT 1,
    CreatedAt                  DATETIME2 NOT NULL DEFAULT GETUTCDATE()
);

-- =============================================
-- EMPLOYEE DOCUMENTS
-- Stores employee uploaded documents
-- =============================================
CREATE TABLE EmployeeDocument (
    DocumentId                 BIGINT PRIMARY KEY IDENTITY(1,1),
    EmployeeId                 BIGINT NOT NULL,
    DocumentTypeId             BIGINT NOT NULL,
    FileName                   NVARCHAR(500),
    OriginalFileName           NVARCHAR(500),
    FileExtension              NVARCHAR(20),
    MimeType                   NVARCHAR(100),
    FileSizeInBytes            BIGINT,
    FileUrl                    NVARCHAR(1000),
    DocumentNumber             NVARCHAR(200),
    IssuedDate                 DATE NULL,
    ExpiryDate                 DATE NULL,
    Remarks                    NVARCHAR(1000),
    UploadedAt                 DATETIME2 NOT NULL DEFAULT GETUTCDATE(),
    IsVerified                 BIT NOT NULL DEFAULT 0,
    VerifiedByEmployeeId       BIGINT NULL,
    VerifiedAt                 DATETIME2 NULL,
    IsActive                   BIT NOT NULL DEFAULT 1,

    CONSTRAINT FK_EmployeeDocument_Employee
        FOREIGN KEY (EmployeeId)
        REFERENCES Employee(EmployeeId),

    CONSTRAINT FK_EmployeeDocument_DocumentType
        FOREIGN KEY (DocumentTypeId)
        REFERENCES DocumentType(DocumentTypeId),

    CONSTRAINT FK_EmployeeDocument_VerifiedBy
        FOREIGN KEY (VerifiedByEmployeeId)
        REFERENCES Employee(EmployeeId)
);



-- =============================================
-- APPROVAL TYPE MASTER
-- Stores workflow approval types
-- =============================================
CREATE TABLE ApprovalType (
    ApprovalTypeId             BIGINT PRIMARY KEY IDENTITY(1,1),
    ApprovalTypeName           NVARCHAR(200) NOT NULL UNIQUE,
    Description                NVARCHAR(1000),
    IsActive                   BIT NOT NULL DEFAULT 1
);

-- =============================================
-- APPROVAL HIERARCHY
-- Stores approval workflows
-- =============================================
CREATE TABLE ApprovalHierarchy (
    ApprovalHierarchyId        BIGINT PRIMARY KEY IDENTITY(1,1),
    ApprovalTypeId             BIGINT NOT NULL,
    RequesterEmployeeId        BIGINT NOT NULL,
    ApproverEmployeeId         BIGINT NOT NULL,
    PriorityOrder              INT NOT NULL,
    EffectiveFrom              DATE,
    EffectiveTo                DATE,
    IsActive                   BIT NOT NULL DEFAULT 1,

    CONSTRAINT FK_AH_ApprovalType
        FOREIGN KEY (ApprovalTypeId)
        REFERENCES ApprovalType(ApprovalTypeId),

    CONSTRAINT FK_AH_Requester
        FOREIGN KEY (RequesterEmployeeId)
        REFERENCES Employee(EmployeeId),

    CONSTRAINT FK_AH_Approver
        FOREIGN KEY (ApproverEmployeeId)
        REFERENCES Employee(EmployeeId)
);
